var SHEET_ID = '10EUU_Cb8yy1-CSCVw2jOeSl_OCR-l4K0MeMQxcGEub8';
var HEADERS = ['管理番号','到着日','顧客名','連絡先','商品名','ブランド','顧客区分','種別','依頼内容','追加詳細','見積','請求','納期','ステータス','引渡日','完了日','備考','削除フラグ','写真'];

function orderToRow(o) {
  return [
    o.id||'', o.arr||'', o.cli||'', o.tel||'', o.item||'', o.brand||'',
    o.ctype||'', o.svc||'', o.req||'', o.det||'',
    o.est||0, o.chg||0, o.due||'', o.st||'', o.dlv||'', o.completedAt||'',
    o.note||'', false, JSON.stringify(o.photos||[])
  ];
}

function rowToOrder(row) {
  return {
    id: String(row[0]),
    arr: String(row[1]||''),
    cli: String(row[2]||''),
    tel: String(row[3]||''),
    item: String(row[4]||''),
    brand: String(row[5]||''),
    ctype: String(row[6]||'一般'),
    svc: String(row[7]||''),
    req: String(row[8]||''),
    det: String(row[9]||''),
    est: Number(row[10])||0,
    chg: Number(row[11])||0,
    due: String(row[12]||''),
    st: String(row[13]||'受付中'),
    dlv: String(row[14]||''),
    completedAt: String(row[15]||''),
    note: String(row[16]||''),
    photos: (function(){ try{ return JSON.parse(row[18]||'[]'); }catch(e){return [];} })()
  };
}

function getSheet(ss) {
  var s = ss.getSheetByName('顧客管理');
  if (!s) {
    s = ss.insertSheet('顧客管理');
    s.appendRow(HEADERS);
    s.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold');
  }
  return s;
}

function findRow(sheet, id) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;
  var ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(id)) return i + 2;
  }
  return -1;
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = getSheet(ss);
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return ok_data([]);
    var rows = sheet.getRange(2, 1, lastRow - 1, 19).getValues();
    var orders = rows
      .filter(function(row) { return row[0] && !row[17]; })
      .map(rowToOrder);
    return ok_data(orders);
  } catch(err) {
    return ok_data([]);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = getSheet(ss);

    if (data.action === 'add_order') {
      sheet.appendRow(orderToRow(data.data.order));
      return ok();
    }

    if (data.action === 'update_order') {
      var o = data.data.order;
      var rowNum = findRow(sheet, o.id);
      if (rowNum > 0) {
        var row = orderToRow(o);
        sheet.getRange(rowNum, 1, 1, 17).setValues([row.slice(0, 17)]);
        sheet.getRange(rowNum, 19).setValue(row[18]);
      }
      return ok();
    }

    if (data.action === 'delete_order') {
      var rowNum = findRow(sheet, data.data.id);
      if (rowNum > 0) sheet.getRange(rowNum, 18).setValue(true);
      return ok();
    }

    return ok();
  } catch(err) {
    return ContentService.createTextOutput('{"status":"error"}').setMimeType(ContentService.MimeType.JSON);
  }
}

function ok() {
  return ContentService.createTextOutput('{"status":"ok"}').setMimeType(ContentService.MimeType.JSON);
}

function ok_data(orders) {
  return ContentService.createTextOutput(JSON.stringify({orders: orders})).setMimeType(ContentService.MimeType.JSON);
}
